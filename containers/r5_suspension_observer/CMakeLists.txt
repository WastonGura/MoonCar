cmake_minimum_required(VERSION 3.16)
project(mooncar_suspension_observer LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
add_library(mooncar-suspension-observer SHARED SuspensionObserver.cc)
target_link_libraries(mooncar-suspension-observer PRIVATE gz-sim8::gz-sim8)
target_compile_features(mooncar-suspension-observer PRIVATE cxx_std_17)
target_compile_options(mooncar-suspension-observer PRIVATE -Wall -Wextra -Werror)
find_package(DART REQUIRED)
add_library(mooncar-native-spring-probe SHARED NativeSpringProbe.cc)
target_link_libraries(mooncar-native-spring-probe PRIVATE dart dl)
target_compile_features(mooncar-native-spring-probe PRIVATE cxx_std_17)
target_compile_options(mooncar-native-spring-probe PRIVATE -Wall -Wextra -Werror)
add_library(mooncar-idle-slip-policy SHARED IdleSlipPolicy.cc)
target_link_libraries(mooncar-idle-slip-policy PRIVATE gz-sim8::gz-sim8)
target_compile_features(mooncar-idle-slip-policy PRIVATE cxx_std_17)
target_compile_options(mooncar-idle-slip-policy PRIVATE -Wall -Wextra -Werror)
