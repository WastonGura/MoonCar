cmake_minimum_required(VERSION 3.16)
project(mooncar_turn_runtime LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
foreach(pair IN ITEMS "mooncar-axle-allocation|AxleAllocation.cc" "mooncar-turn-slip-control|TurnSlipControl.cc")
  string(REPLACE "|" ";" parts "${pair}")
  list(GET parts 0 name)
  list(GET parts 1 source)
  add_library(${name} SHARED ${source})
  target_link_libraries(${name} PRIVATE gz-sim8::gz-sim8)
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
endforeach()
add_library(mooncar-adaptive-axle-allocation SHARED AdaptiveAxleAllocation.cc)
target_link_libraries(mooncar-adaptive-axle-allocation PRIVATE gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
target_compile_features(mooncar-adaptive-axle-allocation PRIVATE cxx_std_17)
target_compile_options(mooncar-adaptive-axle-allocation PRIVATE -Wall -Wextra -Werror)
