cmake_minimum_required(VERSION 3.16)
project(mooncar_turn_slip_control LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
add_library(mooncar-turn-slip-control SHARED TurnSlipControl.cc)
target_link_libraries(mooncar-turn-slip-control PRIVATE gz-sim8::gz-sim8)
target_compile_features(mooncar-turn-slip-control PRIVATE cxx_std_17)
target_compile_options(mooncar-turn-slip-control PRIVATE -Wall -Wextra -Werror)
add_executable(gate-policy-test GatePolicyTest.cc)
target_compile_features(gate-policy-test PRIVATE cxx_std_17)
target_compile_options(gate-policy-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
enable_testing()
add_test(NAME gate-policy COMMAND gate-policy-test)
