cmake_minimum_required(VERSION 3.16)
project(mooncar_integral_contact LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(DART REQUIRED COMPONENTS collision-ode)
add_library(mooncar-integral-contact SHARED IntegralContact.cc)
target_link_libraries(mooncar-integral-contact PRIVATE gz-sim8::gz-sim8)
add_executable(r5-integral-contact-bench IntegralBench.cc)
target_link_libraries(r5-integral-contact-bench PRIVATE dart dart-collision-ode)
foreach(target mooncar-integral-contact r5-integral-contact-bench)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()

add_executable(r5-integral-math IntegralMath.cc)
target_compile_features(r5-integral-math PRIVATE cxx_std_17)
target_compile_options(r5-integral-math PRIVATE -Wall -Wextra -Werror)
