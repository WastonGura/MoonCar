cmake_minimum_required(VERSION 3.16)
project(mooncar_slip_observer LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_library(mooncar-origin-observer SHARED SlipObserver.cc)
target_link_libraries(mooncar-origin-observer PRIVATE gz-sim8::gz-sim8 gz-transport13::core gz-msgs10::gz-msgs10)
target_compile_features(mooncar-origin-observer PRIVATE cxx_std_17)
target_compile_options(mooncar-origin-observer PRIVATE -Wall -Wextra -Werror)
