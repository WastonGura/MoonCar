cmake_minimum_required(VERSION 3.16)
project(mooncar_image_delivery LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(mooncar-ros-image-observer RosImageObserver.cc)
ament_target_dependencies(mooncar-ros-image-observer rclcpp sensor_msgs)
target_compile_features(mooncar-ros-image-observer PRIVATE cxx_std_17)
target_compile_options(mooncar-ros-image-observer PRIVATE -Wall -Wextra)
