cmake_minimum_required(VERSION 3.16)
project(mooncar_nonlinear_contact LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(DART REQUIRED COMPONENTS collision-ode)
add_library(mooncar-nonlinear-contact SHARED NonlinearContact.cc)
target_link_libraries(mooncar-nonlinear-contact PRIVATE gz-sim8::gz-sim8)
add_executable(r5-nonlinear-contact-bench NonlinearBench.cc)
target_link_libraries(r5-nonlinear-contact-bench PRIVATE dart dart-collision-ode)
foreach(target mooncar-nonlinear-contact r5-nonlinear-contact-bench)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
